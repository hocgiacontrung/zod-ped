# ============================================================================
# requirements.txt — env for the ZOD pedestrian intent/trajectory pipeline.
# Conda env `zod-iac`, Python 3.10, CUDA 12.x, RTX 4080.
#
#   pip install -r requirements.txt
#
# Tiers (top to bottom = install order; order matters):
#   [A] core pipeline    — Steps 1-3 + world-frame validation. numpy/scipy/shapely only.
#   [B] notebooks & viz  — exploration + results notebooks (optional for headless).
#   [C] 2D + frustum     — 2D detector recall (scripts/eval_detector2d_recall.py) and the
#                          frustum POC (scripts/frustum_poc.py): YOLO box -> ZOD projection
#                          -> LiDAR depth. This is the retained Step 2 measurement path.
#
# NOTE: off-the-shelf 3D detectors (PointPillars / OpenPCDet) were dropped after the
# bring-up gates (see docs/EXPERIMENTS_LOG.md; code recoverable at the
# `experiments/3d-detectors` tag). Their stack — spconv, numba, easydict, gdown, etc. —
# is intentionally NOT listed here. Any future heavy detector (Detectron2, SAM4D) should
# get its OWN conda env so `zod-iac` stays clean and conflict-free.
# ============================================================================

# --- [0] NumPy pin — install FIRST, everything below assumes it -----------------
# torch 2.2.0+cu121 is built against NumPy 1.x; NumPy 2.x -> "_ARRAY_API not found"
# and breaks all torch<->numpy interop. This single pin is the #1 environment blocker.
numpy==1.26.4

# --- [A] Core pipeline (the only third-party imports in src/ + scripts/) ---------
scipy==1.15.3
shapely==2.1.2
zod==0.8.0          # ZOD devkit — low-level utils only (NOT the ZodSequences loader)
pyyaml==6.0.3       # configs/*.yaml

# --- [B] Notebooks & visualization ----------------------------------------------
# opencv 4.11 is the last line that still allows numpy<2 on py3.10 (4.13 forces
# numpy>=2, which conflicts with the pin above). Imports fine at runtime.
matplotlib==3.10.9
opencv-python==4.11.0.86
notebook==7.5.6
tqdm==4.67.3

# --- [C] 2D detector + frustum (retained Step 2 measurement path) ----------------
# torch already in zod-iac; pinned for reproducibility. To reinstall:
#   pip install torch==2.2.0 torchvision==0.17.0 --index-url https://download.pytorch.org/whl/cu121
torch==2.2.0
torchvision==0.17.0
ultralytics==8.4.69     # YOLO11x 2D detector (frustum front-end)
