# MANUAL STEPS — NOT pip-installable. Needed ONLY for the detector (tier C) run.
# (Verified working on zod-iac, 2026-06-17. Full notes: memory/detector_env_buildout.)
#
# 1. CUDA toolkit + a compatible compiler (nvcc is absent by default):
#       conda install -c nvidia -c conda-forge cuda-toolkit=12.1 gxx=12
#    NOTE: the solver may give CUDA 12.9 — fine (major-compatible with torch 12.1).
#    Headers land in $CONDA_PREFIX/targets/x86_64-linux/include (NOT .../include),
#    so before building OpenPCDet export:
#       export CUDA_HOME="$CONDA_PREFIX"
#       export CC="$CONDA_PREFIX/bin/x86_64-conda-linux-gnu-gcc"
#       export CXX="$CONDA_PREFIX/bin/x86_64-conda-linux-gnu-g++"
#       export CPATH="$CONDA_PREFIX/targets/x86_64-linux/include:$CPATH"
#       export LIBRARY_PATH="$CONDA_PREFIX/targets/x86_64-linux/lib:$CONDA_PREFIX/lib:$LIBRARY_PATH"
#       export TORCH_CUDA_ARCH_LIST="8.9"   # RTX 4080 (Ada)
#
# 2. OpenPCDet (pcdet) — build from source, NOT a wheel:
#       git clone https://github.com/open-mmlab/OpenPCDet.git ~/OpenPCDet
#       cd ~/OpenPCDet && pip install -e . --no-build-isolation
#    (plain `setup.py develop` fails — build isolation hides torch.)
#    Two local patches to the clone were needed (see memory/detector_env_buildout):
#      - make the argo2 dataset import optional in pcdet/datasets/__init__.py
#      - replace removed numpy aliases np.int/np.float/np.bool in 5 pcdet files
#
# 3. Pedestrian-capable checkpoint (nuScenes PP-MultiHead):
#       cd ~/OpenPCDet && mkdir -p checkpoints && cd checkpoints
#       gdown 1p-501mTWsq0G9RzroTWSXreIMyTUUpBM -O cbgs_pp_multihead_nds5823_updated.pth
#    Run the gate FROM ~/OpenPCDet/tools (so the cfg's relative _BASE_CONFIG_ resolves):
#       cd ~/OpenPCDet/tools && python <repo>/scripts/eval_detector_recall.py \
#         --detector pointpillars \
#         --pp-cfg  cfgs/nuscenes_models/cbgs_pp_multihead.yaml \
#         --pp-ckpt ~/OpenPCDet/checkpoints/cbgs_pp_multihead_nds5823_updated.pth \
#         --score-thresh 0.1   # intensity is fed RAW 0-255 by default (nuScenes convention)
